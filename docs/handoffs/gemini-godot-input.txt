CarLoop / Godot native implementation handoff / 2026-11-17

아래는 실제 저장소 파일의 텍스트 스냅샷이다. 웹의 GameCore를 기준으로 Windows에서 바로 실행 가능한 Godot 4 2D 버전을 구현해줘.
목표는 iOS 우선 출시, 이후 Android이며, 무료 다운로드 + IAP/IAA, 자체 서버 없는 구조다.
이번에는 기본 플레이만 이식한다. 실제 광고/결제/게임센터 SDK와 HubGame, Three.js, 무기/미션 기능은 추가하지 않는다.

작업 전 확인:
- 웹 GameCore의 updateProgress, validSnapshot, checkDeadlock을 읽었는지 확인해라.
- project.godot의 main_scene 경로가 가리키는 Main 씬은 현재 없다. 실행 가능한 씬을 제공해라.
- 파일을 읽을 수 없으면 추정하지 말고 필요한 원문을 요청해라.

구현 조건:
1. 순수 GDScript 코어와 Node 기반 UI/저장 컨트롤러를 분리한다. 코어는 DOM/Node/파일/SDK를 참조하지 않는다.
2. res://data/levels.json을 그대로 사용하고 기존 웹의 규칙과 두 레벨을 보존한다.
3. lane은 수집 노드다. 잠긴 차량은 수집하지 않는다. 블록은 수집 가능한 차량의 입구를 통과할 때만 수집된다.
4. 사출/순환 이동/자동 수집/완성/전체 클리어/막힘/1x·2x/Undo20개/Shuffle표시순서/Reset을 구현한다.
5. 트랙이 가득 차고 수집이 불가능하면 차량 내부 블록이 남아 있어도 막힘이다.
6. 120Hz 고정 시뮬레이션, delta 상한0.1초, 포커스 복귀 시 누적 시간 초기화. 색상별 블록은 생성/유실되지 않는다.
7. Main.tscn/Main.gd를 제공한다. 외부 자산 없이 세로 화면과 안전 영역, 터치/마우스를 지원한다. UI 전체를 매 프레임 다시 만들지 않는다.
8. user:// 로컬 저장: 버전/레벨/상태 검증, 정상 백업/안전한 파일 교체, 손상 복구, 중단/포커스 상실 시 저장. 매 프레임 저장하지 않는다.
9. 클리어 전환은 컨트롤러가 한 번 예약하고 Undo/Reset/재시작 때 취소한다. 코어는 자체 레벨 전환을 하지 않는다.
10. autoload/class_name 충돌, 잘못된 프로젝트 설정과 기존 스캐폴드의 중복 규칙을 정리한다.
11. 헤드리스 검증 스크립트: 두 레벨 클리어, 60/120Hz 동일 속도, 2배속, 블록 보존, Undo, 막힘과 잘못된 저장을 검사한다.
12. 두 레벨의 기준 해법은 A4회->B4회->C4회이며 각 사출 후 수집을 충분히 기다린다.

const LEVELS = JSON.parse(document.getElementById('levelData').textContent);
const clone = value => JSON.parse(JSON.stringify(value));

class GameCore {
  constructor(levels) {
    this.levels = levels;
    this.blockCounter = 0;
    this.accumulator = 0;
    this.baseSpeed = 3;
    this.initLevel(0);
  }

  initLevel(levelIdx, speed = 1) {
    this.state = {
      levelIdx, speed, cars: clone(this.levels[levelIdx].cars),
      track: [], cleared: [], history: []
    };
    this.accumulator = 0;
    this.markCompleted();
  }

  markCompleted() {
    for (const car of this.state.cars) {
      if (car.blocks.length === car.capacity && car.blocks.every(color => color === car.color)
          && !this.state.cleared.includes(car.id)) this.state.cleared.push(car.id);
    }
  }

  validSnapshot(snapshot, levelIdx) {
    const level = this.levels[levelIdx];
    if (!snapshot || snapshot.levelIdx !== levelIdx || ![1, 2].includes(snapshot.speed)
        || !Array.isArray(snapshot.cars) || snapshot.cars.length !== level.cars.length
        || !Array.isArray(snapshot.track) || snapshot.track.length > level.trackCapacity
        || !Array.isArray(snapshot.cleared)) return false;
    const seen = new Set();
    const counts = Object.fromEntries(level.cars.map(car => [car.color, 0]));
    for (const car of snapshot.cars) {
      if (!car || seen.has(car.id)) return false;
      const template = level.cars.find(candidate => candidate.id === car.id);
      if (!template || ['color', 'capacity', 'lane', 'locked'].some(key => car[key] !== template[key])
          || !Array.isArray(car.blocks) || car.blocks.length > car.capacity) return false;
      seen.add(car.id);
      for (const color of car.blocks) {
        if (!Object.hasOwn(counts, color)) return false;
        counts[color]++;
      }
      const complete = car.blocks.length === car.capacity && car.blocks.every(color => color === car.color);
      if (snapshot.cleared.includes(car.id) !== complete) return false;
    }
    const ids = new Set();
    for (const item of snapshot.track) {
      if (!item || !Number.isSafeInteger(item.id) || item.id <= 0 || ids.has(item.id)
          || !Object.hasOwn(counts, item.color) || !Number.isFinite(item.progress)
          || item.progress < 0 || item.progress >= level.trackNodes.length) return false;
      ids.add(item.id);
      counts[item.color]++;
    }
    if (new Set(snapshot.cleared).size !== snapshot.cleared.length
        || snapshot.cleared.some(id => !seen.has(id))) return false;
    return Object.entries(counts).every(([color, count]) =>
      count === level.cars.filter(car => car.color === color).reduce((sum, car) => sum + car.capacity, 0));
  }

  loadState(saved) {
    if (!saved || !Number.isInteger(saved.levelIdx) || !this.levels[saved.levelIdx]
        || !this.validSnapshot(saved, saved.levelIdx) || !Array.isArray(saved.history)
        || saved.history.length > 20
        || !saved.history.every(snapshot => this.validSnapshot(snapshot, saved.levelIdx))) return false;
    this.state = clone(saved);
    this.accumulator = 0;
    this.blockCounter = Math.max(0, ...[saved, ...saved.history].flatMap(snapshot => snapshot.track.map(item => item.id)));
    return true;
  }

  snapshot() {
    const snapshot = clone({ ...this.state, history: undefined });
    this.state.history.push(snapshot);
    if (this.state.history.length > 20) this.state.history.shift();
  }

  undo() {
    const previous = this.state.history.pop();
    if (!previous) return false;
    this.state = { ...previous, history: this.state.history, speed: this.state.speed };
    this.accumulator = 0;
    return true;
  }

  shuffle() {
    if (this.isCleared()) return false;
    this.snapshot();
    for (let index = this.state.cars.length - 1; index > 0; index--) {
      const target = Math.floor(Math.random() * (index + 1));
      [this.state.cars[index], this.state.cars[target]] = [this.state.cars[target], this.state.cars[index]];
    }
    return true;
  }

  launchCar(carId) {
    if (this.isCleared()) return 'clearing';
    const car = this.state.cars.find(candidate => candidate.id === carId);
    if (!car || car.locked || this.state.cleared.includes(car.id) || !car.blocks.length) return 'invalid';
    if (this.state.track.length >= this.levels[this.state.levelIdx].trackCapacity) return 'full';
    this.snapshot();
    this.state.track.push({ id: ++this.blockCounter, color: car.blocks.shift(), progress: 0 });
    return 'launched';
  }

  canCollect(car, color) {
    return !car.locked && !this.state.cleared.includes(car.id) && car.color === color
      && car.blocks.length < car.capacity && car.blocks.every(block => block === color);
  }

  updateProgress(deltaTime) {
    if (!Number.isFinite(deltaTime) || deltaTime <= 0 || this.isCleared()) return { moved: false, collected: false };
    this.accumulator += Math.min(deltaTime, 0.1);
    const step = 1 / 120;
    const level = this.levels[this.state.levelIdx];
    let moved = false;
    let collected = false;
    while (this.accumulator + 1e-10 >= step) {
      this.accumulator = Math.max(0, this.accumulator - step);
      const distance = this.baseSpeed * this.state.speed * step;
      for (let index = this.state.track.length - 1; index >= 0; index--) {
        const item = this.state.track[index];
        const start = item.progress;
        item.progress = (start + distance) % level.trackNodes.length;
        moved = true;
        const car = this.state.cars.find(candidate => {
          const dock = candidate.lane % level.trackNodes.length;
          const toDock = (dock - start + level.trackNodes.length) % level.trackNodes.length;
          return this.canCollect(candidate, item.color) && toDock > 1e-10 && toDock <= distance + 1e-10;
        });
        if (car) {
          car.blocks.push(item.color);
          this.state.track.splice(index, 1);
          collected = true;
          this.markCompleted();
        }
      }
    }
    return { moved, collected };
  }

  checkDeadlock() {
    const level = this.levels[this.state.levelIdx];
    return this.state.track.length >= level.trackCapacity
      && !this.state.track.some(item => this.state.cars.some(car => this.canCollect(car, item.color)));
  }

  isCleared() {
    return this.state.cleared.length === this.state.cars.length;
  }
}

function save(force = false) {
  const now = performance.now();
  if (!dirty || (!force && now - lastSaveTime < 1000)) return;
  lastSaveTime = now;
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify({
      version: 3, levels: LEVEL_SIGNATURE, state: core.state
    }));
    dirty = false;
    el.saveStatus.textContent = '진행 상황 자동 저장 중';
  } catch {
    el.saveStatus.textContent = '저장할 수 없습니다. 현재 창에서는 계속 플레이할 수 있습니다.';
  }
}

function cancelClear() {
  revision++;
  if (clearTimer !== null) clearTimeout(clearTimer);
  clearTimer = null;
}

function getNodePosition(index) {
  const nodes = LEVELS[core.state.levelIdx].trackNodes;
  const maxX = Math.max(...nodes.map(node => node[0])) || 1;
  const maxY = Math.max(...nodes.map(node => node[1])) || 1;
  const node = nodes[index % nodes.length];
  return { x: 6 + node[0] / maxX * 88, y: 12 + node[1] / maxY * 72 };
}

    button.className = 'car' + (car.locked ? ' locked' : '') + (complete ? ' cleared' : '');
    button.disabled = car.locked || complete || !car.blocks.length;
    button.setAttribute('aria-label', car.id + ' 차량, 목표 ' + car.color + ', 블록 ' + car.blocks.length + '개');
    button.innerHTML = '<div class="car-head"><strong>' + car.id + ' · target ' + car.color
      + '</strong><span class="tiny">' + (complete ? 'CLEARED' : car.blocks.length + '/' + car.capacity)
      + '</span></div><div class="slots">' + (car.blocks.map(color =>
        '<span class="pill ' + color + '">' + color[0].toUpperCase() + '</span>').join('')
        || '<span class="tiny">empty</span>') + '</div>';
    button.onclick = () => {
      const result = core.launchCar(car.id);
      if (result === 'launched') {
        statusMessage = car.id + '에서 블록을 사출했습니다.';
        dirty = true;
        save(true);
        renderCars();
        renderDynamic();
        updateStatus();
      } else if (result === 'full') {
        setStatus('트랙이 가득 찼습니다. 수집을 기다리거나 Undo를 사용하세요.');
      }
    };
    el.cars.appendChild(button);
  }
  el.undoBtn.disabled = !core.state.history.length;
}

function renderStatic() {
  const level = LEVELS[core.state.levelIdx];
  el.levelName.textContent = level.name + ' (' + level.id + ')';
  el.speedBtn.textContent = core.state.speed + 'x';
  el.trackCanvas.replaceChildren();
  blockElements.clear();
  level.trackNodes.forEach((node, index) => {
    const position = getNodePosition(index);
    const element = document.createElement('div');
    element.className = 'node-bg';
    element.textContent = index + 1;
    const cars = level.cars.filter(car => car.lane % level.trackNodes.length === index);
    if (cars.length) element.textContent = cars.map(car => car.id).join('/');
    element.title = cars.length ? cars.map(car => car.id + ' 수집 입구').join(', ') : '트랙 ' + (index + 1);
    element.style.left = position.x + '%';
    element.style.top = position.y + '%';
    el.trackCanvas.appendChild(element);
  });
  renderCars();
  renderDynamic();
}

function renderDynamic() {
  const level = LEVELS[core.state.levelIdx];
  el.capacityLabel.textContent = core.state.track.length + ' / ' + level.trackCapacity;
  const active = new Set(core.state.track.map(item => item.id));
  for (const [id, element] of blockElements) {
    if (!active.has(id)) {
      element.remove();
      blockElements.delete(id);
    }
  }
  for (const item of core.state.track) {
    let element = blockElements.get(item.id);
    if (!element) {
      element = document.createElement('div');
      element.className = 'moving-block ' + item.color;
      element.textContent = item.color[0].toUpperCase();
      blockElements.set(item.id, element);
      el.trackCanvas.appendChild(element);
    }
    const index = Math.floor(item.progress);
    const ratio = item.progress - index;
    const start = getNodePosition(index);
    const end = getNodePosition(index + 1);
    element.style.left = (start.x + (end.x - start.x) * ratio) + '%';
    element.style.top = (start.y + (end.y - start.y) * ratio) + '%';
  }
}

function updateStatus() {
  if (core.isCleared()) {
    setStatus('클리어! 다음 레벨로 넘어갑니다.');
    if (clearTimer === null) {
      const scheduledRevision = revision;
      clearTimer = setTimeout(() => {
        if (scheduledRevision !== revision || !core.isCleared()) return;
        clearTimer = null;
        core.initLevel((core.state.levelIdx + 1) % LEVELS.length, core.state.speed);
        dirty = true;
        statusMessage = '다음 레벨입니다.';
        save(true);
        renderStatic();
        updateStatus();
      }, 1200);
    }
  } else {
    setStatus(core.checkDeadlock() ? '막힘 상태입니다. Undo 또는 Reset을 사용하세요.' : statusMessage);
  }
}

function gameLoop(timestamp) {
  const deltaTime = lastTime === null ? 0 : Math.max(0, (timestamp - lastTime) / 1000);
  lastTime = timestamp;
  const result = core.updateProgress(deltaTime);
  if (result.moved) {
    dirty = true;
    renderDynamic();
  }
  if (result.collected) renderCars();
  save(result.collected);
  updateStatus();
  rafId = requestAnimationFrame(gameLoop);
}

function initGame() {
  cancelClear();
  if (rafId !== null) cancelAnimationFrame(rafId);
  core.initLevel(0);
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (raw) {
      const saved = JSON.parse(raw);
      if (saved.version !== 3 || saved.levels !== LEVEL_SIGNATURE || !core.loadState(saved.state)) {
        statusMessage = '이전 버전 또는 잘못된 저장 데이터입니다. 새 게임을 시작합니다.';
      }
    }
  } catch {
    statusMessage = '저장 데이터를 읽을 수 없어 새 게임을 시작합니다.';
  }
  dirty = true;
  save(true);
  renderStatic();
  updateStatus();
  lastTime = null;
  rafId = requestAnimationFrame(gameLoop);
}

===== FILE: godot-project/scripts/LevelController.gd =====
extends Node
class_name LevelController

var level_data: Dictionary = {}
var cleared_car_ids: Array[String] = []

func load_level(next_level: Dictionary) -> void:
	level_data = next_level.duplicate(true)
	cleared_car_ids.clear()

func get_tappable_car_ids() -> Array[String]:
	var ids: Array[String] = []
	for car in level_data.get("cars", []):
		if bool(car.get("locked", false)):
			continue
		if car.get("blocks", []).is_empty():
			continue
		ids.append(String(car.get("id", "")))
	return ids

func can_collect(car: Dictionary, color: String) -> bool:
	var blocks: Array = car.get("blocks", [])
	if blocks.size() >= int(car.get("capacity", 4)):
		return false
	if blocks.is_empty():
		return true
	for block in blocks:
		if String(block) != color:
			return false
	return String(car.get("color", "")) == color

func collect_to_car(car_id: String, color: String) -> bool:
	for car in level_data.get("cars", []):
		if String(car.get("id", "")) != car_id:
			continue
		if not can_collect(car, color):
			return false
		car["blocks"].append(color)
		if car["blocks"].size() >= int(car.get("capacity", 4)):
			if _is_uniform(car["blocks"], String(car.get("color", ""))):
				cleared_car_ids.append(car_id)
		return true
	return false

func is_cleared() -> bool:
	return cleared_car_ids.size() >= level_data.get("cars", []).size()

func is_deadlocked(track_count: int, track_capacity: int, any_collectable: bool) -> bool:
	var no_space := track_count >= track_capacity
	var no_tap := get_tappable_car_ids().is_empty()
	return no_space and no_tap and not any_collectable

func _is_uniform(blocks: Array, color: String) -> bool:
	for block in blocks:
		if String(block) != color:
			return false
	return true

func setup(track_nodes: Array, track_capacity: int) -> void:
	nodes = track_nodes.duplicate(true)
	capacity = track_capacity
	items.clear()

func can_push() -> bool:
	return items.size() < capacity

func push_block(color: String) -> bool:
	if not can_push():
		return false
	items.append({"color": color, "progress": 0})
	return true

func step_all(delta_units: int = 1) -> void:
	for item in items:
		item["progress"] = int(item.get("progress", 0)) + delta_units

func pop_matching(color: String) -> bool:
	for idx in range(items.size()):
		if String(items[idx]["color"]) == color:
			items.remove_at(idx)
			return true
	return false
